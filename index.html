#!/bin/bash REPO="giantpreston/gdasher" OS=$(uname -s | tr '[:upper:]' '[:lower:]') RAW_ARCH=$(uname -m) case "$RAW_ARCH" in x86_64|amd64) ARCH="x64" ;; aarch64|arm64) ARCH="arm64" ;; *) exit 1 ;; esac case "$OS" in linux*) PLATFORM="linux" ;; darwin*) PLATFORM="macos" ;; *) exit 1 ;; esac FILENAME="gdasher-$PLATFORM-$ARCH" DESKTOP_PATH="$HOME/Desktop" DOWNLOAD_URL=$(curl -s https://api.github.com/repos/$REPO/releases/latest | grep "browser_download_url" | grep "$FILENAME" | cut -d '"' -f 4) if [ -z "$DOWNLOAD_URL" ]; then exit 1 fi curl -L "$DOWNLOAD_URL" -o "$DESKTOP_PATH/$FILENAME" chmod +x "$DESKTOP_PATH/$FILENAME"
