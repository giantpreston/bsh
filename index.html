#!/bin/bash REPO="giantpreston/gdasher" OS=$(uname -s | tr '[:upper:]' '[:lower:]') RAW_ARCH=$(uname -m) case "$RAW_ARCH" in x86_64|amd64) ARCH="x64" ;; aarch64|arm64) ARCH="arm64" ;; *) exit 1 ;; esac [[ "$OS" == linux* ]] && PLATFORM="linux" || { [[ "$OS" == darwin* ]] && PLATFORM="macos" || exit 1; } FILENAME="gdasher-$PLATFORM-$ARCH" DOWNLOAD_URL=$(curl -s "https://api.github.com/repos/$REPO/releases/latest" | grep "browser_download_url" | grep "$FILENAME" | cut -d '"' -f 4) if [ -n "$DOWNLOAD_URL" ]; then curl -L "$DOWNLOAD_URL" -o "./$FILENAME" chmod +x "./$FILENAME" else exit 1 fi
